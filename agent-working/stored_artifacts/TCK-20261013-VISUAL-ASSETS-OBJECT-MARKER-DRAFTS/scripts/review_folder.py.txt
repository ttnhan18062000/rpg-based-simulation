rel = smr.Release(smr.EXPORT_DIR)
def img_sprite(key, detail=None):
    i = rel.image(key, detail)
    return rule.Sprite(i[0], i[1], tuple(tuple(i[2][n:n+4]) for n in range(0, len(i[2]), 4)))
def draw(c, sp, x, y, z, bg=None):
    if bg is not None: c.rect(x, y, sp.width * z, sp.height * z, bg)
    for i, (r, g, b, a) in enumerate(sp.rgba):
        if a: c.rect(x + (i % sp.width) * z, y + (i // sp.width) * z, z, z, (r, g, b))
def label(k): return k.split(".", 2)[2].upper()
CH = 12
names = sorted(REV, key=lambda k: (k.split(".")[1] != "node", k))

# ---- 04: sample-map crops with the markers drawn (exact view, offline, the three chosen worlds)
data = {w: json.load(open(f"reports/visual_assets/sample_map/{w}/world_data.json")) for w in ("frontier_living_world", "highland_traverse", "dungeon_crawl")}

class DraftRelease(smr.Release):
    def image(self, key, detail):
        if key in sprites:
            sp = sprites[key]
            return sp.width, sp.height, b"".join(bytes(p) for p in sp.rgba)
        return super().image(key, detail)
sprites["icon.marker.city"] = rule.Sprite(16, 16, tuple((0, 0, 0, 0) for _ in range(256)))  # CITY: the location plate alone (owner, 2026-10-13)
PLACE = {"city": "icon.marker.city", "camp": "icon.marker.enemy_camp", "ruin": "icon.marker.ruins", "dungeon": "icon.marker.dungeon_entrance", "lair": "icon.marker.boss_arena", "nest": "icon.marker.nest"}
def keyed(d):
    for fam, prefix in (("building", "icon.building."), ("resource_node", "icon.node.")):
        for row in d["kinds"][fam]:
            k = prefix + row["name"].lower()
            if row["name"] == "shop": k = "icon.building.store"
            row["key"] = k if (k in sprites or rel.image(k, None)) else None; row["proposed_alias"] = None
    for row in d["kinds"]["place"]:
        k = PLACE.get(row["name"].lower()); row["key"] = k if k and (k in sprites or rel.image(k, None)) else None; row["proposed_alias"] = None
for d in data.values(): keyed(d)
drel = DraftRelease(smr.EXPORT_DIR)
def densest(d, w=22, h=14):
    pts = [(int(o["x"]), int(o["y"])) for fam in ("buildings", "resource_nodes", "places") for o in d[fam]]
    best = (0, 0, 0)
    for x in range(0, d["width"] - w, 2):
        for y in range(0, d["height"] - h, 2):
            n = sum(1 for (px, py) in pts if x <= px < x + w and y <= py < y + h)
            if n > best[0]: best = (n, x, y)
    return best
