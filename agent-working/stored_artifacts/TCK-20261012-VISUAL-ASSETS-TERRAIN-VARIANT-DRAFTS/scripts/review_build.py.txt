def sample8(png: bytes) -> bytes:
    img = pixels.decode_png(png); assert (img.width, img.height) == (128, 128)
    out = bytearray()
    for y in range(16):
        for x in range(16):
            i = ((y*8)*128 + x*8)*4; out += img.rgba[i:i+4]
    return bytes(out)
class Rel:
    release_id = "review"
    def __init__(self, with_variants: bool):
        self.base = r.Release(r.EXPORT_DIR)
        self.var = {}
        if with_variants:
            for (k, d), iid in INTAKES.items(): self.var[(k, d)] = sample8((Q/iid/"preview.png").read_bytes())
        self.axes = dict(self.base.axes)
        if with_variants:
            for k, vals in AXES.items(): self.axes[k] = {"values": vals, "default": "base"}
    def image(self, key, detail):
        if (key, detail) in self.var: return (16, 16, self.var[(key, detail)])
        if key in AXES and detail == "base": return self.base.image(key, None)
        return self.base.image(key, detail)
    def tile(self, key, x, y):
        axis = self.axes.get(key)
        if axis is None:
            img = self.image(key, None); return (img[2], None) if img else (None, None)
        picked = r.pick_detail(key, x, y, r.DETAIL_SEED, axis["values"])
        for d in (picked, axis["default"]):
            img = self.image(key, d)
            if img: return img[2], d
        return None, picked
    def variants(self, k): return self.base.variants(k)
    def mask(self, k, d): return self.base.mask(k, d)
def world(grid):  # grid of key names -> data dict for draw_world
    names = sorted({n for row in grid for n in row}); flat = [names.index(n) for row in grid for n in row]
    from tools import visual_assets_sample_map as sm
    return {"world_id": "review", "role": "", "width": len(grid[0]), "height": len(grid), "terrain_names": names, "terrain_rle": sm.rle(flat),
            "terrain": [{"name": n, "count": 1, "key": f"terrain.{n}", "near": []} for n in names], "buildings": [], "places": [], "resource_nodes": [],
            "kinds": {"building": [], "place": [], "resource_node": [], "entity": []}, "proposed_aliases": {"label": "x"}}
def png_of(canvas, w, h, scale):
    from PIL import Image
    im = Image.frombytes("RGBA", (w, h), b"".join(bytes(row) for row in canvas)).resize((w*scale, h*scale), Image.NEAREST); return im
def draw_map(grid, rel, scale):
    canvas, stats = r.draw_world(world(grid), rel, "exact"); return png_of(canvas, len(grid[0])*16, len(grid)*16, scale), stats
FONT = ImageFont.load_default()
def labelled(tiles, labels, scale=16, cols=None, title=""):
    cols = cols or len(tiles); pad = 14; lab = 28
    W = cols*(16*scale+pad)+pad; H = (len(tiles)+cols-1)//cols*(16*scale+pad+lab)+pad+24
    im = Image.new("RGB", (W, H), (60, 66, 60)); d = ImageDraw.Draw(im); d.text((pad, 6), title, fill=(235, 235, 220), font=FONT)
    for i, (t, l) in enumerate(zip(tiles, labels)):
        x = pad + (i % cols)*(16*scale+pad); y = 24 + (i//cols)*(16*scale+pad+lab)
        tile = Image.frombytes("RGBA", (16, 16), t).resize((16*scale, 16*scale), Image.NEAREST); im.paste(tile, (x, y), tile)
        d.text((x, y+16*scale+6), l, fill=(235, 235, 220), font=FONT)
    return im
